cmake_minimum_required(VERSION 3.20)
project(CTEAS_P1_Terrel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(GDAL REQUIRED)
add_library(cteas_p1_grid STATIC ../src/geo/terrain.cpp ../src/geo/terrain_grid.cpp)
target_include_directories(cteas_p1_grid PUBLIC ../include)
target_link_libraries(cteas_p1_grid PUBLIC GDAL::GDAL)
if(MSVC)
  target_compile_options(cteas_p1_grid PRIVATE /W4 /WX)
else()
  target_compile_options(cteas_p1_grid PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(cteas_p1_terrain ../src/p1_terrain_cli.cpp)
target_link_libraries(cteas_p1_terrain PRIVATE cteas_p1_grid)
add_executable(cteas_p1_cellind_tests ../tests/p1_cellind_tests.cpp)
target_link_libraries(cteas_p1_cellind_tests PRIVATE cteas_p1_grid)
enable_testing()
add_test(NAME cteas_p1_cellind_scoped COMMAND cteas_p1_cellind_tests)

# Enforce case-grid precision independently from data availability.
add_executable(cteas_grid_origin_precision_tests ${CMAKE_CURRENT_SOURCE_DIR}/../tests/grid_origin_precision_tests.cpp)
target_include_directories(cteas_grid_origin_precision_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../include)
add_test(NAME cteas_grid_origin_three_decimal_km COMMAND cteas_grid_origin_precision_tests)
add_executable(cteas_p1_dynamic_utm_tests ../tests/p1_dynamic_utm_tests.cpp)
target_link_libraries(cteas_p1_dynamic_utm_tests PRIVATE cteas_p1_grid)
add_test(NAME cteas_p1_dynamic_utm_software_only COMMAND cteas_p1_dynamic_utm_tests)
add_executable(cteas_p1_terrain_case ../src/p1_terrain_case_cli.cpp)
target_link_libraries(cteas_p1_terrain_case PRIVATE cteas_p1_grid)
