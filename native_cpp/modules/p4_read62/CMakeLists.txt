cmake_minimum_required(VERSION 3.20)
project(CTEAS_P4_READ62_Scoped LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cteas_p4_tests ../../tests/p4_read62_tests.cpp)
target_include_directories(cteas_p4_tests PRIVATE ../../include)
if(MSVC)
  target_compile_options(cteas_p4_tests PRIVATE /W4 /WX)
else()
  target_compile_options(cteas_p4_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  # Preserve assert-based C++ contract tests in Release (CMake defines NDEBUG).
  target_compile_options(cteas_p4_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME p4_fsl_input_contracts COMMAND cteas_p4_tests)

add_executable(cteas_p4_read62 p4_read62_main.cpp)
target_include_directories(cteas_p4_read62 PRIVATE ../../include)
if(MSVC)
  target_compile_options(cteas_p4_read62 PRIVATE /W4 /WX)
else()
  target_compile_options(cteas_p4_read62 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
